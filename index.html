async function handleImageUpload(event) { const file = event.target.files[0]; if (!file) return; const statusEl = document.getElementById('ocrStatus'); statusEl.innerText = "⏳ AI is scanning and formatting chords..."; try { const { data: { text } } = await Tesseract.recognize(file, 'eng'); // Clean up OCR noise/garbage lines let cleanedLines = text.split('\n') .map(line => line.trim()) .filter(line => line.length > 0 && !/^[^a-zA-Z0-9()\[\]]+$/.test(line)); // Process scanned text to wrap standalone chords in (Chord) format let formattedText = cleanedLines.map(line => { // Regex matching standalone chords like G, C, D7, Am, F#, Bb/F const chordPattern = /\b([A-G][#b]?[m7maj9susdim]*(\/[A-G][#b]?)?)\b/g; // If line contains mostly chords, wrap each chord in parentheses const words = line.split(/\s+/); const chordMatches = line.match(chordPattern) || []; if (chordMatches.length > 0 && chordMatches.length >= words.length / 2) { return line.replace(chordPattern, '($1)'); } // If line has inline chords without brackets, convert them to (Chord) return line.replace(/\[([A-G][#b]?[^\]]*)\]/g, '($1)'); }).join('\n'); document.getElementById('songLyrics').value = formattedText; statusEl.innerText = "✨ Scan complete! Chords formatted to (G) style."; } catch (err) { console.error("OCR Error:", err); statusEl.innerText = "❌ Error reading image. Please type lyrics manually."; } }
